import 'dart:async';
import 'dart:io';
import 'lan_socket_stub.dart';

class _IoLanTransport implements LanSocketTransport {
  final String url;
  WebSocket? _socket;
  StreamSubscription? _sub;
  final StreamController<String> _controller = StreamController<String>.broadcast();

  _IoLanTransport(this.url);

  @override
  Stream<String> get incomingText => _controller.stream;

  @override
  Future<void> connect() async {
    final socket = await WebSocket.connect(url).timeout(const Duration(seconds: 5));
    _socket = socket;
    _sub = socket.listen(
      (event) {
        if (event is String) {
          _controller.add(event);
        }
      },
      onDone: () {
        if (!_controller.isClosed) _controller.close();
      },
      onError: (_) {
        if (!_controller.isClosed) _controller.close();
      },
    );
  }

  @override
  void sendText(String data) {
    try {
      _socket?.add(data);
    } catch (_) {}
  }

  @override
  Future<void> close() async {
    await _sub?.cancel();
    _sub = null;
    try {
      await _socket?.close();
    } catch (_) {}
    _socket = null;
    if (!_controller.isClosed) {
      await _controller.close();
    }
  }
}

LanSocketTransport createLanTransport(String url) => _IoLanTransport(url);
