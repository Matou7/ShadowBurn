import 'dart:async';
import 'dart:html' as html;
import 'lan_socket_stub.dart';

class _WebLanTransport implements LanSocketTransport {
  final String url;
  html.WebSocket? _socket;
  final StreamController<String> _controller = StreamController<String>.broadcast();

  _WebLanTransport(this.url);

  @override
  Stream<String> get incomingText => _controller.stream;

  @override
  Future<void> connect() async {
    final completer = Completer<void>();
    final ws = html.WebSocket(url);
    _socket = ws;

    ws.onOpen.first.then((_) {
      if (!completer.isCompleted) completer.complete();
    });

    ws.onError.first.then((_) {
      if (!completer.isCompleted) {
        completer.completeError(StateError("Échec ouverture WebSocket Web : $url"));
      }
    });

    ws.onMessage.listen((html.MessageEvent event) {
      final data = event.data;
      if (data is String) {
        _controller.add(data);
      }
    });

    ws.onClose.listen((_) {
      if (!_controller.isClosed) {
        _controller.close();
      }
    });

    await completer.future.timeout(const Duration(seconds: 5));
  }

  @override
  void sendText(String data) {
    final ws = _socket;
    if (ws != null && ws.readyState == html.WebSocket.OPEN) {
      ws.send(data);
    }
  }

  @override
  Future<void> close() async {
    try {
      _socket?.close();
    } catch (_) {}
    _socket = null;
    if (!_controller.isClosed) {
      await _controller.close();
    }
  }
}

LanSocketTransport createLanTransport(String url) => _WebLanTransport(url);
